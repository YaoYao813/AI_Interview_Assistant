# 1. 创建虚拟环境
cd "C:\Users\yao\Desktop\Study\项目\AI Interview Assistant"
python -m venv venv

# 2. 激活虚拟环境（Windows PowerShell）
venv\Scripts\Activate.ps1

# 3. 安装依赖
pip install -r requirements.txt

# 4. 创建 .env 文件（复制模板，填入真实 API Key）
copy .env.example .env

# 5. 启动 Redis
docker-compose up -d

# 6. 启动后端（开发模式，Debug=True 时自动 reload）
cd backend
uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
